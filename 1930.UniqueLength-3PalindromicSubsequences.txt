    public int countPalindromicSubsequence(String s) {

        int n = s.length();

        Set<Character> set = new HashSet<>();

        for(int i=0;i<n;i++){
            set.add(s.charAt(i));
        }


        

        int result=0;

        for(char c : set){
            int first=-1;
            int last =-1;

            for(int i=0;i<n;i++){
                if(s.charAt(i)==c){
                    if(first==-1){
                    first=i;
                    }
                    last=i;
                }
            }

            Set<Character> set2 = new HashSet<>();
            for(int mid = first+1;mid<=last-1;mid++){
                set2.add(s.charAt(mid));
            }
            result+= set2.size();

        }

        return result;
